users
  └─ {userId}
      ├─ householdId: [array de strings]
      ├─ isAdmin: boolean
      └─ name: string

households
  └─ {householdId}
      ├─ familyName: string
      ├─ members: { [userId]: boolean }

      ├─ categories (subcoleção)
      │    └─ {categoryId}
      │         └─ name: string

      ├─ types (subcoleção)
      │    └─ {typeId}
      │         ├─ name: string
      │         └─ isIncome: boolean

      ├─ plannedTransactions (subcoleção)
      │    └─ {plannedTransactionId}
      │         ├─ amount: number
      │         ├─ category_id: string
      │         ├─ description: string
      │         ├─ date: timestamp
      │         ├─ supplier: string
      │         ├─ type_id: string
      │         ├─ user_id: string
      │         ├─ installments_current: number
      │         ├─ installments_total: number
      │         ├─ transactionGroupId: string
      │         └─ yearMonth: string

      ├─ transactions (subcoleção)
      │    └─ {transactionId}
      │         ├─ amount: number
      │         ├─ category_id: string
      │         ├─ description: string
      │         ├─ date: timestamp
      │         ├─ supplier: string
      │         ├─ type_id: string
      │         ├─ user_id: string
      │         ├─ installments_current: number
      │         ├─ installments_total: number
      │         ├─ transactionGroupId: string
      │         └─ yearMonth: string

      ├─ budgets (subcoleção)
      │   └─ {year}
      │        └─ categories (subcoleção)
      │             └─ {categoryId}
      │                  └─ types: { [typeId]: { valor: number } }

      ├─ investments (subcoleção)
      │    └─ {investmentId}
      │         ├─ name: string
      │         ├─ startDate: timestamp            
      │         ├─ initialAmount: number
      │         ├─ currentAmount: number
      │         ├─ annualYield: number
      │         ├─ projection5y: number
      │         ├─ projection10y: number
      │         ├─ projection20y: number
      │         ├─ projection30y: number
      │         ├─ allocations: { [goalId]: number } // { "goalA": 0.5, "goalB": 0.5 }
      │
      │         └─ history (subcoleção)
      │              └─ {historyId}
      │                   ├─ date: timestamp
      │                   ├─ value: number
      │                   └─ returnRate: number

      ├─ goals (subcoleção)
      │    └─ {goalId}
      │         ├─ name: string
      │         ├─ startDate: timestamp           
      │         ├─ targetAmount: number
      │         ├─ targetDate: timestamp
      │         ├─ progress: number
      │         ├─ status: string                  // 'active' | 'achieved' | 'exceeded' | 'cancelled'
      │         ├─ expectedReturnRate: number
      │         ├─ linkedInvestments: [array de strings]
      │         └─ surplusAmount: number           // valor excedente acima da meta (se houver)

      └─ investmentAllocations (subcoleção)
           └─ {allocationId}
                ├─ investmentId: string
                ├─ goalId: string
                └─ percentage: number
