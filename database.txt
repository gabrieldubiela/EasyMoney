users
  └─ {userId}
      ├─ householdId: [array de strings]
      ├─ isAdmin: boolean
      └─ name: string

households
  └─ {householdId}
      ├─ familyName: string
      ├─ members: { [userId]: boolean }
      ├─ categories (subcoleção)
      │    └─ {categoryId}
      │         └─ name: string
      ├─ types (subcoleção)
      │    └─ {typeId}
      │         ├─ name: string
      │         └─ isIncome: boolean
      ├─ plannedTransactions (subcoleção)
      │    └─ {plannedTransactionId}
      │         ├─ amount: number
      │         ├─ category_id: string
      │         ├─ description: string
      │         ├─ date: timestamp
      │         ├─ supplier: string
      │         ├─ type_id: string
      │         ├─ user_id: string
      │         ├─ installments_current: number
      │         ├─ installments_total: number
      │         ├─ transactionGroupId: string
      │         └─ yearMonth: string   // ex.: "202510"
      ├─ transactions (subcoleção)
      │    └─ {transactionId}
      │         ├─ amount: number
      │         ├─ category_id: string
      │         ├─ description: string
      │         ├─ date: timestamp
      │         ├─ supplier: string
      │         ├─ type_id: string
      │         ├─ user_id: string
      │         ├─ installments_current: number
      │         ├─ installments_total: number
      │         ├─ transactionGroupId: string
      │         └─ yearMonth: string
      ├─ budgets (subcoleção)
      │   └─ {year}
      │        └─ categories (subcoleção)
      │             └─ {categoryId}
      │                  └─ types: { [typeId]: { valor: number } }
      ├─ alerts (subcoleção)
      │    └─ {alertId}
      │         ├─ alertType: string
      │         ├─ ... (demais campos definidos no app)
      ├─ investments (subcoleção)
      │    └─ {investmentId}
      │         ├─ name: string
      │         ├─ startDate: timestamp            
      │         ├─ initialAmount: number
      │         ├─ currentAmount: number
      │         ├─ annualYield: number
      │         ├─ projection5y/projection10y/...: number
      │         ├─ allocations: { [goalId]: number }
      │         └─ history (subcoleção)
      │              └─ {historyId}
      │                   ├─ date: timestamp
      │                   ├─ value: number
      │                   └─ returnRate: number
      ├─ goals (subcoleção)
      │    └─ {goalId}
      │         ├─ name, startDate, targetAmount, targetDate, progress, status, expectedReturnRate, linkedInvestments, surplusAmount
      ├─ investmentAllocations (subcoleção)
      │    └─ {allocationId}
      │         ├─ investmentId: string
      │         ├─ goalId: string
      │         └─ percentage: number
